\subsection{问题提出}

\begin{frame}{问题提出}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{BCI 系统约束\cite{lotte2007review,lotte2011regularizing}}
            \begin{itemize}
                \item \textbf{数据稀缺}：50--100 试次校准
                \item \textbf{资源受限}：KB--MB 级内存
                \item \textbf{延迟要求}：毫秒级响应
            \end{itemize}
        \end{block}

        \column{0.48\textwidth}
        \begin{block}{时间窗方法对比\cite{mcfarland1997spatial,yang2015time,chen2019dynamic,zhang2021mars}}
            \renewcommand{\arraystretch}{1.1}
            \begin{tabular}{l|l}
                \hline
                \textbf{方法} & \textbf{局限} \\
                \hline
                固定时间窗 & 无法适应个体差异 \\
                滑动窗 & 计算复杂度线性增长 \\
                贝叶斯优化 & 高维参数效率低 \\
                深度强化学习 & 百万参数、黑盒 \\
                \hline
            \end{tabular}
        \end{block}
    \end{columns}

    \vspace{0.2cm}
    \begin{alertblock}{核心矛盾}
        DQN 的资源需求 \textbf{vs} BCI 系统受限现实
    \end{alertblock}
\end{frame}

\begin{frame}{研究问题与方案}
    \begin{alertblock}{研究问题}
        \centering
        是否存在一种方法，能在保持与 DQN 相当性能的同时，\\
        \textbf{降低模型复杂度}、\textbf{提升稳定性}和\textbf{可解释性}？
    \end{alertblock}

    \vspace{0.2cm}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{核心洞察}
            时间窗优化任务的状态空间是\textbf{有限的}：
            \centering\large\textbf{136 个离散状态}
        \end{block}

        \column{0.48\textwidth}
        \begin{alertblock}{研究方案}
            使用\textbf{表格型 Q-Learning}\cite{watkins1992q}替代 DQN
            \begin{itemize}
                \item 维护 Q 表，无需函数近似
                \item $\varepsilon$-贪婪策略
                \item CSP-SVM 准确率为奖励
            \end{itemize}
        \end{alertblock}
    \end{columns}
\end{frame}
