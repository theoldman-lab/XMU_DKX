\begin{frame}{数据集与预处理}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{alertblock}{BCI Competition IV 2a 数据集\cite{tangermann2012review}}
            \footnotesize
            \begin{itemize}
                \item \textbf{被试}：9 名健康被试
                \item \textbf{任务}：4 类运动想象（左手、右手、脚、舌头）
                \item \textbf{范式}：基于 cue，试次$\approx$6 秒，想象阶段 0--4 秒
                \item \textbf{采集}：22 通道 EEG, 250 Hz, 10-20 系统
                \item \textbf{划分}：训练集 (session T) 288 试次，测试集 (session E) 288 试次
            \end{itemize}
        \end{alertblock}

        \column{0.5\textwidth}
        \begin{block}{预处理流程\cite{delorme2004eeglab}}
            \footnotesize
            \begin{enumerate}
                \item \textbf{带通滤波}：8--30 Hz (FIR)\\保留$\mu$(8--13 Hz)和$\beta$(13--30 Hz)节律，捕捉 ERD/ERS 现象\cite{pfurtscheller1999event}
                \item \textbf{陷波滤波}：50 Hz (IIR)\\消除工频干扰
                \item \textbf{平均参考}：所有通道平均值\\提高空间分辨率
                \item \textbf{时间窗截取}：$[t_{\text{start}}, t_{\text{end}}]$\\用于特征提取
            \end{enumerate}
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{特征提取与分类}
    \begin{columns}[T]
        \column{0.55\textwidth}
        \begin{block}{CSP 特征提取\cite{blankertz2008optimizing,mcfarland1997spatial}}
            \footnotesize
            \vspace{0.1cm}
            \textbf{共空间模式 (CSP)}：使不同类别信号方差差异最大化
            
            \vspace{0.15cm}
            \textbf{多类别扩展}（近似对角化）：
            \begin{itemize}
                \item 计算各类别平均协方差矩阵 $\bar{\mathbf{C}} = \frac{1}{C}\sum_{c=1}^{C} \mathbf{C}_c$
                \item 特征分解构造白化矩阵 $\mathbf{W}$，使 $\mathbf{W} \bar{\mathbf{C}} \mathbf{W}^{\text{T}} = \mathbf{I}$
                \item 选择最大/最小特征值对应特征向量作为空间滤波器
            \end{itemize}

            \vspace{0.1cm}
            \textbf{特征向量}：$\mathbf{f}_i = \log \left( \text{var}(\mathbf{W} \mathbf{X}_i^{\text{window}}) \right)$
            
            \vspace{0.1cm}
            \textbf{配置}：4 个 CSP 分量（2 对）
        \end{block}

        \column{0.42\textwidth}
        \begin{alertblock}{分类器设计\cite{lotte2007review}}
            \footnotesize
            \textbf{分类器}：线性核 SVM ($C=1.0$)
            
            \vspace{0.15cm}
            \textbf{特征标准化}：Z-score
            \begin{equation*}
                \mathbf{f}_i \leftarrow \frac{\mathbf{f}_i - \mu_{\mathbf{f}}}{\sigma_{\mathbf{f}}}
            \end{equation*}

            \vspace{0.15cm}
            \textbf{交叉验证}：5 折交叉验证
            \begin{itemize}
                \item 训练集划分为 5 份
                \item 4 份训练，1 份验证
                \item 重复 5 次取平均
            \end{itemize}
        \end{alertblock}
    \end{columns}
\end{frame}

\begin{frame}{对比方法与评估协议}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{block}{10 种对比方法}
            \footnotesize
            \begin{itemize}
                \item \textbf{固定时间窗}（3 种）
                \begin{itemize}
                    \scriptsize
                    \item FL-2s：窗口 2.0 s (5 候选)
                    \item FL-1.5s：窗口 1.5 s (6 候选)
                    \item FL-1s：窗口 1.0 s (7 候选)
                \end{itemize}
                \item \textbf{搜索优化}（2 种）
                \begin{itemize}
                    \scriptsize
                    \item Grid-Search：步长 0.2 s (136 候选)
                    \item Random-Search：50 采样
                \end{itemize}
                \item \textbf{强化学习}（5 种）
                \begin{itemize}
                    \scriptsize
                    \item Standard Q-Learning
                    \item Double Q-Learning
                    \item Dueling Q-Learning
                    \item Dueling Double Q-Learning
                    \item DQN (64+32 全连接网络)
                \end{itemize}
            \end{itemize}
        \end{block}

        \column{0.5\textwidth}
        \begin{alertblock}{评估协议\cite{tangermann2012review,cohen1960coefficient}}
            \footnotesize
            \textbf{重复实验}：每名被试 5 次重复 (种子 0--4)，报告均值 $\pm$ 标准差
            
            \textbf{评价指标}：
            \begin{itemize}
                \item 准确率 (Accuracy)
                \item Cohen's Kappa (一致性系数)
                \item F1 分数 (宏平均)
                \item 标准差 (Std Dev)、变异系数 (CV)
            \end{itemize}

            \textbf{统计检验}：配对样本 $t$ 检验 ($\alpha=0.05$) + Cohen's $d$ 效应量
            \begin{itemize}
                \item $|d|<0.2$ 小效应，$0.2\leq|d|<0.5$ 中效应，$|d|\geq0.5$ 大效应
            \end{itemize}
        \end{alertblock}
    \end{columns}
\end{frame}

\begin{frame}{基线方法性能对比分析}
    \begin{alertblock}{子图说明}
        \footnotesize
        \begin{itemize}
            \item \textbf{分类准确率}：误差线表示标准差，搜索策略（61.9\%--63.3\%）显著优于固定窗口（约 45\%）
            \item \textbf{准确率提升百分比}：相对于 Fixed-2s，Grid-Search 和 Random-Search 分别实现 +18.3\% 和 +16.9\%，Fixed-1s 仅 +0.4\%
            \item \textbf{Cohen's Kappa}：搜索策略（0.492--0.511）中等一致性，优于固定窗口（0.27）
            \item \textbf{F1 分数}：搜索策略（0.611--0.625）较固定窗口（0.43）提升约 44\%
        \end{itemize}
    \end{alertblock}

    \begin{exampleblock}{结果分析}
        \footnotesize
        \textbf{性能提升}：搜索优化方法显著优于固定窗口策略。Grid-Search 准确率达 63.35\%，较 Fixed-2s 提升 18.3\%（\textbf{相对提升 40.6\%}）。Kappa 系数从 0.27 提升至 0.51，F1 分数从 0.43 提升至 0.63。这表明\textbf{自适应时间窗口能有效捕捉最优信号片段}，克服固定窗口无法适应 EEG 信号动态变化的局限。
        
        \textbf{固定窗口局限}：三种固定窗口长度性能几乎相同（45.05\%--45.47\%），表明\textbf{简单调整窗口长度无法解决固定窗口的固有局限}，自适应策略是必然选择。
    \end{exampleblock}
\end{frame}

        \textbf{搜索策略比较}：Grid-Search 略优于 Random-Search（+1.4\%），但差异微小（< 2.3\%）。考虑到 Random-Search 的计算效率优势，\textbf{在实时 BCI 系统中可能更具实用价值}。
