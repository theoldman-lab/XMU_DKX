{\footnotesize

\subsection{研究总结}

\begin{frame}{研究背景与核心问题}
    \begin{block}{MI-BCI 应用前景}
        \begin{itemize}
            \item 康复医疗：运动功能障碍患者重建运动功能
            \item 人机交互：新型控制方式
            \item 神经工程：脑信号解码与理解
        \end{itemize}
    \end{block}

    \vspace{0.2cm}
    \begin{alertblock}{关键挑战：时间窗选择}
        \begin{itemize}
            \item ERD 动态特性存在显著个体间差异
            \item EEG 信号非平稳、低信噪比
            \item 在线 BCI 系统需毫秒级响应
        \end{itemize}
    \end{alertblock}

    \vspace{0.2cm}
    \begin{exampleblock}{核心研究问题}
        如何在保持性能的前提下，实现轻量化、可解释的时间窗优化？
    \end{exampleblock}
\end{frame}

\begin{frame}{现有方法局限性分析}
    \begin{table}[h]
        \centering
        \renewcommand{\arraystretch}{1.3}
        \begin{tabular}{l|l}
            \hline
            \textbf{方法} & \textbf{局限性} \\
            \hline
            固定时间窗 & 无法适应个体间差异 \\
            滑动窗 & 计算复杂度线性增长 \\
            贝叶斯优化 & 高维参数空间效率低 \\
            深度强化学习 & 百万参数、训练不稳定、黑盒决策 \\
            \hline
        \end{tabular}
    \end{table}

    \vspace{0.3cm}
    \begin{alertblock}{核心矛盾}
        DQN 对大量训练数据和高计算资源的依赖 \\
        \textbf{vs} \\
        BCI 系统数据稀缺、设备资源受限的现实
    \end{alertblock}
\end{frame}

\begin{frame}{研究方法：表格型 Q-Learning}
    \begin{block}{问题形式化}
        \begin{itemize}
            \item \textbf{状态空间}：136 个离散时间窗状态
            \item \textbf{动作空间}：4 个边界调整动作
            \item \textbf{奖励函数}：CSP-SVM 交叉验证准确率
            \item \textbf{状态转移}：确定性过程
        \end{itemize}
    \end{block}

    \vspace{0.2cm}
    \begin{alertblock}{算法设计}
        \begin{itemize}
            \item 维护 Q 表，无需函数近似
            \item $\varepsilon$-贪婪策略平衡探索与利用
            \item 时序差分学习迭代优化
        \end{itemize}
    \end{alertblock}

    \vspace{0.2cm}
    \begin{exampleblock}{核心优势}
        轻量化 $\cdot$ 收敛性保证 $\cdot$ 可解释性强
    \end{exampleblock}
\end{frame}

\begin{frame}{实验结果与性能对比}
    \begin{center}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{lcc}
        \hline
        \textbf{指标} & \textbf{Table Q} & \textbf{DQN} \\
        \hline
        准确率 & 62.68\% & 62.70\% \\
        统计检验 & \multicolumn{2}{c}{$p = 0.981$} \\
        训练时间 & \textbf{快 3 倍} & 基准 \\
        推理延迟 & \textbf{低 1000 倍} & 基准 \\
        内存占用 & \textbf{低 125 倍} & 基准 \\
        收敛性 & \checkmark 理论保证 & $\times$ 无保证 \\
        可解释性 & \checkmark 完全可解释 & $\times$ 黑盒 \\
        \hline
    \end{tabular}
    \end{center}

    \vspace{0.3cm}
    \begin{alertblock}{核心结论}
        表格型 Q-Learning 在保持与 DQN 相当性能的同时，显著降低计算资源需求
    \end{alertblock}
\end{frame}

\begin{frame}{主要理论贡献}
    \begin{alertblock}{（1）轻量化 BCI 设计新范式}
        \begin{itemize}
            \item 将深度强化学习简化为表格型强化学习
            \item 实现轻量化与高性能的统一
            \item 为资源受限场景下的 BCI 系统优化提供新思路
        \end{itemize}
    \end{alertblock}

    \vspace{0.2cm}
    \begin{exampleblock}{（2）算法选择准则}
        \begin{itemize}
            \item \textbf{适用边界}：状态空间 $< 10^4$ 且状态 - 动作对可充分访问
            \item \textbf{优势}：收敛保证、样本效率高、可解释性强
            \item \textbf{推广价值}：其他状态空间有限的 BCI 优化任务
        \end{itemize}
    \end{exampleblock}
\end{frame}

\begin{frame}{主要实践贡献}
    \begin{alertblock}{（3）可解释性 BCI 系统实践}
        \begin{itemize}
            \item 提供可直接可视化的 Q 值表
            \item 追踪智能体学习轨迹
            \item 支撑临床转化应用
        \end{itemize}
    \end{alertblock}

    \vspace{0.2cm}
    \begin{exampleblock}{（4）实验验证体系}
        \begin{itemize}
            \item 10 种对比方法系统性评估
            \item 配对样本$t$检验 + 效应量分析
            \item 被试级别性能详细对比
        \end{itemize}
    \end{exampleblock}
\end{frame}

\subsection{研究局限性}

\begin{frame}{研究局限性（1/3）}
    \begin{alertblock}{（1）验证范围的局限}
        \begin{itemize}
            \item \textbf{数据集}：仅 BCI Competition IV 2a 标准数据集
            \item \textbf{被试群体}：9 名健康被试，未包含患者
            \item \textbf{验证场景}：离线分析，未在线闭环验证
            \item \textbf{统计效力}：样本规模有限
        \end{itemize}
    \end{alertblock}

    \vspace{0.2cm}
    \begin{exampleblock}{（2）方法设计的局限}
        \begin{itemize}
            \item \textbf{离散化步长}：0.2 秒，可能无法定位全局最优
            \item \textbf{奖励设计}：通过 CSP-SVM 准确率间接构建
            \item \textbf{干扰因素}：CSP 特征质量、SVM 参数等
        \end{itemize}
    \end{exampleblock}
\end{frame}

\begin{frame}{研究局限性（2/3）}
    \begin{alertblock}{（3）Q-Learning 变体研究不足}
        \begin{itemize}
            \item 仅对比 4 种变体（Standard, Double, Dueling, Dueling Double）
            \item 未深入分析 Double Q-Learning 性能略低的内在机制
            \item 未探索其他改进变体：
            \begin{itemize}
                \item Prioritized Experience Replay
                \item Noisy Nets
                \item Distributional RL
            \end{itemize}
            \item 未对 Q 值表演化过程进行深入分析
        \end{itemize}
    \end{alertblock}
\end{frame}

\begin{frame}{研究局限性（3/3）}
    \begin{alertblock}{（4）通用性的局限}
        \begin{itemize}
            \item \textbf{分类器}：仅 CSP-SVM，未验证深度学习模型
            \item \textbf{BCI 范式}：仅运动想象，未验证 SSVEP、P300
            \item \textbf{嵌入式测试}：缺乏 ARM 架构单片机实际运行测试
        \end{itemize}
    \end{alertblock}

    \vspace{0.2cm}
    \begin{exampleblock}{（5）分析维度的局限}
        \begin{itemize}
            \item 理论复杂度分析为主
            \item 缺乏实际功耗和运行时间基准测试
        \end{itemize}
    \end{exampleblock}
\end{frame}

\subsection{未来工作展望}

\begin{frame}{未来工作（1/2）}
    \begin{alertblock}{（1）开展在线和临床验证}
        \begin{itemize}
            \item \textbf{在线实验}：移植到嵌入式便携式 BCI 系统
            \item \textbf{评估指标}：实时反馈场景下的响应速度和用户体验
            \item \textbf{临床验证}：与康复机构合作，在运动功能障碍患者群体中检验
            \item \textbf{预期价值}：为实际应用价值提供更有力证据
        \end{itemize}
    \end{alertblock}

    \vspace{0.2cm}
    \begin{exampleblock}{（2）优化算法效率}
        \begin{itemize}
            \item \textbf{快速奖励估计}：基于元学习或缓存机制
            \item \textbf{目标}：离线训练时间从分钟级缩短到秒级
            \item \textbf{更精细离散化}：探索 0.1 秒步长或结合连续优化
            \item \textbf{预期收益}：提升时间窗选择精度
        \end{itemize}
    \end{exampleblock}
\end{frame}

\begin{frame}{未来工作（2/2）}
    \begin{alertblock}{（3）探索算法泛化能力}
        \begin{itemize}
            \item \textbf{与其他模型结合}：EEGNet、Shallow FBCSP 等
            \item \textbf{空间层面优化}：
            \begin{itemize}
                \item 通道选择：基于强化学习选择最优 EEG 通道子集
                \item CSP 空间滤波器优化：联合优化分量数和参数
                \item 时空联合优化：统一建模为 MDP
            \end{itemize}
        \end{itemize}
    \end{alertblock}

    \vspace{0.2cm}
    \begin{exampleblock}{（4）研究迁移学习策略}
        \begin{itemize}
            \item \textbf{核心思路}：从不同被试的最优 Q 表中提取通用先验知识
            \item \textbf{应用场景}：新用户仅提供 10--20 个试次
            \item \textbf{预期收益}：大幅缩短系统校准时间
        \end{itemize}
    \end{exampleblock}
\end{frame}

\subsection{本章小结}

\begin{frame}{本章小结}
    \begin{block}{研究总结}
        \begin{itemize}
            \item 提出表格型 Q-Learning 轻量化时间窗优化方法
            \item 性能与 DQN 无显著差异 ($p = 0.981$)
            \item 实现 3 倍训练加速、1000 倍推理延迟降低、125 倍内存占用减少
            \item 四大贡献：轻量化范式、算法选择准则、可解释性实践、实验验证体系
        \end{itemize}
    \end{block}
